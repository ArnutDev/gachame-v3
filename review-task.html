เอกสารอธิบายระบบสุ่ม

บันทึกสรุปการทำงานและอธิบายโค้ดระบบสุ่มกาชา (Gacha Engine Randomization)

อ้างอิงจากโครงสร้างสถาปัตยกรรม Gacha Engine ของโครงการ ในไฟล์ src/engine/gachaEngine.ts ระบบสุ่มใช้กลไกการสุ่มตามอัตราถ่วงน้ำหนัก (Weighted Random Selection Algorithm) แบ่งออกเป็น 2 ขั้นตอนหลักดังนี้:

ขั้นตอนที่ 1: การสุ่มเลือกระดับความยากง่าย (Stage 1: Select Rarity)

ระบบจะสุ่มเลือกระดับความยากง่าย (เช่น 8★ Ultra, 8★ Normal, 7★ Ultra, 7★ Normal) จากตารางความน่าจะเป็นที่กำหนดไว้ในแบนเนอร์ โดยมีหลักการทำงานดังนี้:

export function selectRarity(
  rarityRates: Record<string, number>,
  randomFn: () => number = Math.random
): string {
  const rand = randomFn() * 100; // 1. แปลงค่าสุ่มจาก 0.0 - 1.0 ให้เป็น 0 - 100%
  let cumulative = 0;

  const keys = Object.keys(rarityRates).sort(); // 2. เรียงคีย์เพื่อลำดับการเช็คที่แน่นอน
  for (const key of keys) {
    const rate = rarityRates[key] || 0;
    cumulative += rate; // 3. สะสมเรทสุ่มไปเรื่อยๆ (Cumulative Sum)
    if (rand < cumulative) { // 4. หากค่าสุ่มตกอยู่ในช่วงสะสม ให้ส่งคืนระดับนั้นทันที
      return key;
    }
  }
  return keys[keys.length - 1];
}

อธิบายการทำงาน: ตัวแปร rand จะดึงค่าทศนิยมแบบสุ่มแล้วคูณด้วย 100 เพื่อสุ่มตัวเลขระหว่าง 0 ถึง 100 จากนั้นสะสมเรทไปทีละช่อง (เช่น 70% -> 92% -> 97% -> 100%) เมื่อ rand มีค่าน้อยกว่าเรทสะสม ณ ตำแหน่งใด แสดงว่าผลลัพธ์การสุ่มตกอยู่ในช่องระดับความยากง่ายตัวนั้น

ขั้นตอนที่ 2: การสุ่มเลือกไอเทมย่อย (Stage 2: Select Item)

เมื่อได้ระดับระดับความยากง่ายจากการสุ่มขั้นแรกแล้ว (เช่น ได้ 8★ Normal ซึ่งมีอัตราสุ่มรวมในแบนเนอร์เท่ากับ 5.0%) ระบบจะคำนวณอัตราสุ่มย่อยของไอเทมแต่ละตัวในกลุ่มนั้น และใช้กลไกถ่วงน้ำหนักย่อยในการเจาะจงสุ่มเลือกตัวละครหรืออุปกรณ์:

2.1 การคำนวณการกระจายความน่าจะเป็น (Probability Calculation)

ก่อนเลือกไอเทม ระบบจะหาอัตราสุ่มของไอเทมแต่ละตัวโดยอ้างอิงจากฟังก์ชัน calculateItemProbabilities:

  • ตัวละครแนะนำ (Featured Items): จะได้รับอัตราความน่าจะเป็นที่ระบุตรง ๆ ไว้ในคอนฟิกแบนเนอร์ (เช่น Clark โดนกำหนดเรทสเปกไว้ที่ 0.88%)
  • ตัวละครทั่วไป (Non-featured Items): อัตราความน่าจะเป็นที่เหลือจากการหักลบ Featured Items จะถูกนำมาเฉลี่ยแบ่งเท่า ๆ กัน ให้กับตัวละครทั่วไปทุกตัวในระดับความยากง่ายนั้นๆ
  • หากไม่มีตัวละครทั่วไป: อัตราความน่าจะเป็นส่วนที่เหลือทั้งหมดจะถูกแบ่งเฉลี่ยทับกลับลงไปให้กับ Featured Items ทุกตัวในระดับนั้น

2.2 การสุ่มเจาะจงไอเทม (Weighted Item Selection)

export function selectItem<T extends Ranger | Gear>(
  rarityPool: T[],
  itemRates: Map<string, number>,
  rarityRate: number,
  randomFn: () => number = Math.random
): T {
  // ... จัดการกรณีพูลมี 0 หรือ 1 ชิ้น ...
  const rand = randomFn() * rarityRate; // สุ่มตัวเลขตั้งแต่ 0 ถึงเรทสะสมของระดับความยากง่ายนั้นๆ (เช่น 0 - 5.0%)
  let cumulative = 0;

  for (const item of rarityPool) {
    const rate = itemRates.get(item.id) || 0;
    cumulative += rate;
    if (rand < cumulative) {
      return item;
    }
  }
  return rarityPool[rarityPool.length - 1];
}

อธิบายการทำงาน: ฟังก์ชัน selectItem ทำการคูณค่าสุ่มกับ rarityRate ของระดับความยากง่ายปัจจุบัน เพื่อให้ขอบเขตค่าสุ่ม rand เหมาะสมกับระดับนั้น ๆ จากนั้นลูปบวกสะสมเรทของไอเทมทีละชิ้น และเลือกชิ้นที่ค่าสุ่มตกลงในสัดส่วนน้ำหนักที่กำหนดไว้

จุดเด่นของการสุ่มสถาปัตยกรรมนี้ (Engine Features)

  • รองรับการสลับฟังก์ชันสุ่ม (Injectable PRNG): ตัวแปร randomFn ถูกกำหนดให้สามารถรับฟังก์ชันสุ่มใดๆ ก็ได้จากภายนอก ซึ่งโดยปกติจะรันโดยใช้ Math.random ของเบราว์เซอร์ แต่สามารถแทนที่ด้วยฟังก์ชันคำนวณเลขสุ่มเทียม (Pseudo-Random Number Generator: PRNG) ที่ฟิกเมล็ดพันธุ์ (Seed) ได้ เพื่อใช้ในกรณีทดสอบความเสถียรและจำลองค่าสุ่มให้ออกมาเหมือนกันทุกรอบ (Deterministic Testing)
  • ไม่มีระบบ Pity หรือการล็อกเรท: โครงสร้างเป็นไปตามกฎความน่าจะเป็นบริสุทธิ์ (Generic Weighted Selection) ไม่มีการแทรกแซงหรือปรับปรุงเรทสุ่มแบบไดนามิกใดๆ ในตัวโค้ด
